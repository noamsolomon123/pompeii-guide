**Tech Stack:** Vanilla ES modules, Web Audio, `speechSynthesis` (he-IL) as fallback, `getUserMedia`, `navigator.geolocation`, OpenRouter chat completions with `image_url` for vision, `node --test`.

```js
// www/lib/stops.js
// Route order and neighbours. `enabledOptionals` = { [id]: true } for optional stops the user switched on.
export function routeOrder(stops, enabledOptionals = {}) {
  return stops.filter(s => !s.optional || enabledOptionals[s.id]).sort((a, b) => a.order - b.order);
}
export function nextStop(stops, currentId, enabledOptionals = {}) {
  const r = routeOrder(stops, enabledOptionals);
  const i = r.findIndex(s => s.id === currentId);
  if (i === -1) { const cur = byId(stops, currentId); return cur ? (r.find(s => s.order > cur.order) ?? null) : null; }
  return r[i + 1] ?? null;
}
export function byId(stops, id) { return stops.find(s => s.id === id); }
export async function loadStops(url = './stops.json') { const r = await fetch(url); if (!r.ok) throw new Error(`stops ${r.status}`); return r.json(); }
```

```js
// www/lib/geo.js
export function haversineM([lat1, lon1], [lat2, lon2]) {
  const R = 6371e3, r = (x) => x * Math.PI / 180;
  const dφ = r(lat2 - lat1), dλ = r(lon2 - lon1);
  const a = Math.sin(dφ / 2) ** 2 + Math.cos(r(lat1)) * Math.cos(r(lat2)) * Math.sin(dλ / 2) ** 2;
  return 2 * R * Math.asin(Math.sqrt(a));
}
// → [{ stop, distanceM }] nearest first, only stops with gps, at most n.
export function rankByGps(stops, pos, n = 3) {
  return stops.filter(s => Array.isArray(s.gps)).map(s => ({ stop: s, distanceM: haversineM(s.gps, pos) }))
    .sort((a, b) => a.distanceM - b.distanceM).slice(0, n);
}
```

```js
// www/lib/vision.js
import { OR, headers } from './voice/openrouter-messages.js';

export function buildVisionRequest({ model, candidates, imageDataUrl }) {
  const list = candidates.map(c => `${c.id}: ${c.name_en} (${(c.cues || []).join(', ')})`).join('\n');
  const text = `This photo was taken in Pompeii near one of these places:\n${list}\nAnswer with exactly one id from the list, or "none" if it is clearly none of them. No other words.`;
  return { model, messages: [{ role: 'user', content: [{ type: 'text', text }, { type: 'image_url', image_url: { url: imageDataUrl } }] }], max_tokens: 20 };
}
export function parseVisionAnswer(answer, candidates) {
  const a = (answer || '').trim().toLowerCase().replace(/[."']/g, '');
  const hit = candidates.find(c => c.id === a || c.name_en.toLowerCase() === a);
  return hit ? hit.id : null;
}
// Online path. Returns candidate id or null. Throws on network/API failure (caller falls back to GPS).
export async function identifyByVision({ apiKey, model, candidates, imageDataUrl }) {
  const res = await fetch(OR.chat, { method: 'POST', headers: headers(apiKey), body: JSON.stringify(buildVisionRequest({ model, candidates, imageDataUrl })) });
  if (!res.ok) throw new Error(`vision ${res.status}`);
  const j = await res.json();
  return parseVisionAnswer(j.choices?.[0]?.message?.content, candidates);
}
```

```js
// www/lib/narrator.js
export class Narrator {
  constructor({ filePlayer, ttsPlayer }) { this.filePlayer = filePlayer; this.ttsPlayer = ttsPlayer; this.active = null; this.state = 'idle'; this.lastSpoken = ''; }
  async speak(stop, layer) {
    const text = stop[layer] || '';
    this.lastSpoken = text; this.state = 'playing';
    try {
      this.active = this.filePlayer;
      await this.filePlayer.play(`./audio/${stop.id}-${layer}.mp3`);
    } catch (e) {
      if (e?.code !== 'ENOENT') { this.state = 'idle'; this.active = null; throw e; }
      this.active = this.ttsPlayer;
      await this.ttsPlayer.play(text);
    } finally { if (this.state !== 'stopped') this.state = 'idle'; this.active = null; }
  }
  pause() { if (this.state === 'playing') { this.active?.pause(); this.state = 'paused'; } }
  resume() { if (this.state === 'paused') { this.active?.resume(); this.state = 'playing'; } }
  stop() { this.active?.stop(); this.state = 'idle'; }
}

export class AudioFilePlayer {
  constructor() { this.el = null; }
  play(src) {
    return new Promise((resolve, reject) => {
      const el = new Audio(src); this.el = el;
      el.onended = () => resolve();
      el.onerror = () => { const e = new Error(`audio missing ${src}`); e.code = 'ENOENT'; reject(e); };
      el.play().catch(reject);
    });
  }
  pause() { this.el?.pause(); } resume() { this.el?.play(); } stop() { if (this.el) { this.el.pause(); this.el.src = ''; this.el.onended?.(); } }
}
```

```js
  captureFrame() {
    const v = this.video, max = 1024, k = Math.min(1, max / Math.max(v.videoWidth, v.videoHeight));
    const c = document.createElement('canvas'); c.width = Math.round(v.videoWidth * k); c.height = Math.round(v.videoHeight * k);
    c.getContext('2d').drawImage(v, 0, 0, c.width, c.height);
    return c.toDataURL('image/jpeg', 0.8);
  }
}
```

```js
// Voice barge-in: mic RMS above START for 2 consecutive 100 ms chunks → onSpeech(). Level-only; audio chunks are discarded.
import { MicCapture } from './audio-io.js';
const START = 0.12;
export class BargeIn {
  constructor(onSpeech) { this.onSpeech = onSpeech; this.mic = null; this.hot = 0; this.armed = false; }
  async start() { this.mic = new MicCapture({ targetRate: 16000, onChunk: () => {}, onLevel: (l) => this._level(l) }); await this.mic.start(); }
  arm() { this.armed = true; this.hot = 0; } disarm() { this.armed = false; }
  _level(l) { if (!this.armed) return; if (l > START) { if (++this.hot >= 2) { this.armed = false; this.onSpeech(); } } else this.hot = 0; }
  stop() { this.mic?.stop(); this.mic = null; }
}
```
