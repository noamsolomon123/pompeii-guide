# P1 Bake-off Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A single local web page that lets the user hear the same Hebrew tour-guide persona on OpenAI Realtime and Gemini Live (live voice, barge-in) and compare three TTS renders of one 30-second script, so he can pick the voice backend by ear.

**Architecture:** Static page `bakeoff/index.html` + plain ES modules in `bakeoff/lib/`. Pure helpers (PCM/base64/WAV, message builders) are unit-tested with Node's built-in test runner. Browser-only modules (WebRTC, WebSocket, AudioWorklet) are thin and manually tested. No bundler, no framework, no server. API keys live in `localStorage`.

**Tech Stack:** Vanilla JS ES modules, Web Audio API (AudioWorklet), WebRTC (OpenAI), WebSocket (Gemini), `node --test` (Node ≥ 20), Python `http.server` for local hosting.

**Spec:** `docs/superpowers/specs/2026-09-10-pompeii-guide-design.md` §3 P1.

// bakeoff/lib/pcm.js
// Pure PCM helpers. No DOM, no Web Audio. Works in Node and browser.

export function floatToInt16(f32) {
  const out = new Int16Array(f32.length);
  for (let i = 0; i < f32.length; i++) {
    const s = Math.max(-1, Math.min(1, f32[i]));
    out[i] = s < 0 ? Math.round(s * 32768) : Math.round(s * 32767);
  }
  return out;
}

function bytesToBase64(bytes) {
  if (typeof btoa === 'function') {
    let bin = '';
    for (let i = 0; i < bytes.length; i += 0x8000) {
      bin += String.fromCharCode.apply(null, bytes.subarray(i, i + 0x8000));
    }
    return btoa(bin);
  }
  return Buffer.from(bytes).toString('base64');
}

function base64ToBytes(b64) {
  if (typeof atob === 'function') {
    const bin = atob(b64);
    const bytes = new Uint8Array(bin.length);
    for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
    return bytes;
  }
  return new Uint8Array(Buffer.from(b64, 'base64'));
}

export function int16ToBase64(i16) {
  return bytesToBase64(new Uint8Array(i16.buffer, i16.byteOffset, i16.byteLength));
}

export function base64ToInt16(b64) {
  const bytes = base64ToBytes(b64);
  const even = bytes.byteLength - (bytes.byteLength % 2);
  return new Int16Array(bytes.buffer.slice(bytes.byteOffset, bytes.byteOffset + even));
}

// Integer-ratio box-filter downsample (48k->16k = 3:1, 48k->24k = 2:1).
export function downsample(f32, fromRate, toRate) {
  const ratio = fromRate / toRate;
  if (!Number.isInteger(ratio)) throw new Error(`non-integer ratio ${ratio}`);
  const n = Math.floor(f32.length / ratio);
  const out = new Float32Array(n);
  for (let i = 0; i < n; i++) {
    let acc = 0;
    for (let k = 0; k < ratio; k++) acc += f32[i * ratio + k];
    out[i] = acc / ratio;
  }
  return out;
}

export function wavFromInt16(i16, sampleRate) {
  const dataBytes = i16.byteLength;
  const buf = new ArrayBuffer(44 + dataBytes);
  const dv = new DataView(buf);
  const str = (o, s) => { for (let i = 0; i < s.length; i++) dv.setUint8(o + i, s.charCodeAt(i)); };
  str(0, 'RIFF'); dv.setUint32(4, 36 + dataBytes, true); str(8, 'WAVE');
  str(12, 'fmt '); dv.setUint32(16, 16, true); dv.setUint16(20, 1, true); dv.setUint16(22, 1, true);
  dv.setUint32(24, sampleRate, true); dv.setUint32(28, sampleRate * 2, true);
  dv.setUint16(32, 2, true); dv.setUint16(34, 16, true);
  str(36, 'data'); dv.setUint32(40, dataBytes, true);
  new Int16Array(buf, 44).set(i16);
  return buf;
}
```

// bakeoff/lib/gemini-messages.js
// Pure builders/parsers for the Gemini Live (BidiGenerateContent) protocol.

export function wsUrl(apiKey) {
  return `wss://generativelanguage.googleapis.com/ws/google.ai.generativelanguage.v1beta.GenerativeService.BidiGenerateContent?key=${apiKey}`;
}

export function buildSetup({ model, voice, systemPrompt }) {
  return {
    setup: {
      model: `models/${model}`,
      generationConfig: {
        responseModalities: ['AUDIO'],
        speechConfig: { voiceConfig: { prebuiltVoiceConfig: { voiceName: voice } } }
      },
      systemInstruction: { parts: [{ text: systemPrompt }] },
      inputAudioTranscription: {},
      outputAudioTranscription: {},
      realtimeInputConfig: { automaticActivityDetection: {} }
    }
  };
}

export function buildAudioChunk(base64Pcm16) {
  return { realtimeInput: { audio: { data: base64Pcm16, mimeType: 'audio/pcm;rate=16000' } } };
}

export function buildTextTurn(text) {
  return { clientContent: { turns: [{ role: 'user', parts: [{ text }] }], turnComplete: true } };
}

export function parseServerMessage(msg) {
  const out = { setupComplete: false, audio: [], interrupted: false, turnComplete: false, userText: null, modelText: null };
  if (msg.setupComplete) out.setupComplete = true;
  const sc = msg.serverContent;
  if (sc) {
    for (const p of sc.modelTurn?.parts ?? []) {
      if (p.inlineData?.data && (p.inlineData.mimeType || '').startsWith('audio/')) out.audio.push(p.inlineData.data);
    }
    out.interrupted = !!sc.interrupted;
    out.turnComplete = !!sc.turnComplete;
    out.userText = sc.inputTranscription?.text ?? null;
    out.modelText = sc.outputTranscription?.text ?? null;
  }
  return out;
}
```

// bakeoff/lib/openai-messages.js
// Pure builders/parsers for the OpenAI Realtime API (GA event shapes).

export function callsUrl(model) {
  return `https://api.openai.com/v1/realtime/calls?model=${model}`;
}

export function buildSessionUpdate({ instructions, voice }) {
  return {
    type: 'session.update',
    session: {
      type: 'realtime',
      instructions,
      audio: {
        input: {
          turn_detection: { type: 'server_vad', threshold: 0.5, prefix_padding_ms: 300, silence_duration_ms: 500 },
          transcription: { model: 'gpt-4o-mini-transcribe', language: 'he' }
        },
        output: { voice }
      }
    }
  };
}

export function buildTextTurn(text) {
  return { type: 'conversation.item.create', item: { type: 'message', role: 'user', content: [{ type: 'input_text', text }] } };
}

export function buildResponseCreate() {
  return { type: 'response.create' };
}

export function parseServerEvent(ev) {
  const out = { userText: null, modelText: null, error: null, speechStarted: false };
  switch (ev.type) {
    case 'conversation.item.input_audio_transcription.completed': out.userText = ev.transcript ?? null; break;
    case 'response.output_audio_transcript.done': out.modelText = ev.transcript ?? null; break;
    case 'input_audio_buffer.speech_started': out.speechStarted = true; break;
    case 'error': out.error = ev.error?.message ?? 'unknown error'; break;
  }
  return out;
}

import { floatToInt16, int16ToBase64, base64ToInt16, downsample } from './pcm.js';

// Mic capture: emits base64 PCM16 chunks at targetRate (16000), ~100ms each.
export class MicCapture {
  constructor({ targetRate = 16000, onChunk, onLevel }) {
    this.targetRate = targetRate; this.onChunk = onChunk; this.onLevel = onLevel;
    this.ctx = null; this.stream = null; this.node = null;
  }
  async start() {
    this.stream = await navigator.mediaDevices.getUserMedia({
      audio: { channelCount: 1, echoCancellation: true, noiseSuppression: true, autoGainControl: true }
    });
    this.ctx = new AudioContext({ sampleRate: 48000 });
    await this.ctx.audioWorklet.addModule(new URL('./worklet-capture.js', import.meta.url));
    const src = this.ctx.createMediaStreamSource(this.stream);
    this.node = new AudioWorkletNode(this.ctx, 'capture-processor');
    src.connect(this.node);
    let acc = new Float32Array(0);
    const chunkLen = Math.round(this.ctx.sampleRate * 0.1); // 100 ms
    this.node.port.onmessage = (e) => {
      const f = e.data;
      const merged = new Float32Array(acc.length + f.length); merged.set(acc); merged.set(f, acc.length); acc = merged;
      if (acc.length >= chunkLen) {
        const frame = acc.subarray(0, chunkLen); acc = acc.subarray(chunkLen).slice(0);
        let sum = 0; for (let i = 0; i < frame.length; i++) sum += frame[i] * frame[i];
        this.onLevel?.(Math.min(1, Math.sqrt(sum / frame.length) * 4));
        const ds = downsample(frame, this.ctx.sampleRate, this.targetRate);
        this.onChunk?.(int16ToBase64(floatToInt16(ds)));
      }
    };
  }
  stop() {
    this.node?.disconnect(); this.stream?.getTracks().forEach(t => t.stop()); this.ctx?.close();
    this.node = this.stream = this.ctx = null;
  }
}

// PCM player: queues base64 PCM16 chunks at a fixed rate, plays gaplessly, flush() on interruption.
export class PcmPlayer {
  constructor({ sampleRate = 24000 }) {
    this.sampleRate = sampleRate; this.ctx = null; this.nextTime = 0; this.sources = new Set();
  }
  ensure() { if (!this.ctx) this.ctx = new AudioContext({ sampleRate: this.sampleRate }); if (this.ctx.state === 'suspended') this.ctx.resume(); }
  enqueue(b64) {
    this.ensure();
    const i16 = base64ToInt16(b64);
    const f32 = new Float32Array(i16.length);
    for (let i = 0; i < i16.length; i++) f32[i] = i16[i] / 32768;
    const buf = this.ctx.createBuffer(1, f32.length, this.sampleRate);
    buf.copyToChannel(f32, 0);
    const src = this.ctx.createBufferSource(); src.buffer = buf; src.connect(this.ctx.destination);
    const t = Math.max(this.ctx.currentTime + 0.02, this.nextTime);
    src.start(t); this.nextTime = t + buf.duration;
    this.sources.add(src); src.onended = () => this.sources.delete(src);
  }
  flush() { for (const s of this.sources) { try { s.stop(); } catch {} } this.sources.clear(); this.nextTime = 0; }
  close() { this.flush(); this.ctx?.close(); this.ctx = null; }
}

import { base64ToInt16, wavFromInt16 } from './pcm.js';

export async function geminiTts({ apiKey, text, voice = 'Kore', model = 'gemini-2.5-flash-preview-tts' }) {
  const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`, {
    method: 'POST', headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      contents: [{ parts: [{ text: `קרא בעברית ישראלית טבעית, בקול של מדריך טיולים חם: ${text}` }] }],
      generationConfig: { responseModalities: ['AUDIO'], speechConfig: { voiceConfig: { prebuiltVoiceConfig: { voiceName: voice } } } }
    })
  });
  if (!res.ok) throw new Error(`gemini tts ${res.status}: ${(await res.text()).slice(0, 200)}`);
  const j = await res.json();
  const part = j.candidates?.[0]?.content?.parts?.find(p => p.inlineData);
  if (!part) throw new Error('gemini tts: no audio part');
  const rate = Number((/rate=(\d+)/.exec(part.inlineData.mimeType) || [])[1] || 24000);
  const wav = wavFromInt16(base64ToInt16(part.inlineData.data), rate);
  return URL.createObjectURL(new Blob([wav], { type: 'audio/wav' }));
}

export async function openaiTts({ apiKey, text, voice = 'alloy', model = 'gpt-4o-mini-tts' }) {
  const res = await fetch('https://api.openai.com/v1/audio/speech', {
    method: 'POST', headers: { Authorization: `Bearer ${apiKey}`, 'Content-Type': 'application/json' },
    body: JSON.stringify({ model, voice, input: text, instructions: 'Speak natural Israeli Hebrew with a native accent, warm tour-guide tone.', response_format: 'mp3' })
  });
  if (!res.ok) throw new Error(`openai tts ${res.status}: ${(await res.text()).slice(0, 200)}`);
  return URL.createObjectURL(await res.blob());
}

export async function elevenTts({ apiKey, text, voiceId = 'JBFqnCBsd6RMkjVDRZzb', model = 'eleven_multilingual_v2' }) {
  const res = await fetch(`https://api.elevenlabs.io/v1/text-to-speech/${voiceId}?output_format=mp3_44100_128`, {
    method: 'POST', headers: { 'xi-api-key': apiKey, 'Content-Type': 'application/json' },
    body: JSON.stringify({ text, model_id: model })
  });
  if (!res.ok) throw new Error(`elevenlabs tts ${res.status}: ${(await res.text()).slice(0, 200)}`);
  return URL.createObjectURL(await res.blob());
}
